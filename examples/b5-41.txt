# Skript 5.41: Fleißiger Biber mit 6 Nichtendzuständen (Tabelle S. 66)
# Eingabealphabet {1}: der Lauf startet auf leerem Band, also mit dem Eingabewort ε (5.40).
tm Biber6 {
  alphabet 1
  blank □
  states A B C D E F Z
  start A
  accept Z
  A, □ -> B, 1, R
  A, 1 -> E, □, L
  B, □ -> C, 1, L
  B, 1 -> A, □, R
  C, □ -> D, 1, L
  C, 1 -> C, □, R
  D, □ -> E, 1, L
  D, 1 -> F, □, L
  E, □ -> A, 1, L
  E, 1 -> C, 1, L
  F, □ -> E, 1, L
  F, 1 -> Z, 1, R
  layout A@0,0 B@240,0 C@480,0 D@720,0 E@0,150 F@240,150 Z@480,150
}
