# Eigenes Beispiel (nicht im Skript): fleißiger Biber mit 2 Nichtendzuständen, Bestwert 4 Einsen in 6 Schritten
# Eingabealphabet {1}: der Lauf startet auf leerem Band, also mit dem Eingabewort ε.
tm Biber2 {
  alphabet 1
  blank □
  states A B Z
  start A
  accept Z
  A, □ -> B, 1, R
  A, 1 -> B, 1, L
  B, □ -> A, 1, L
  B, 1 -> Z, 1, R
  layout A@0,0 B@240,0 Z@480,0
}
