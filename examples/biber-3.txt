# Eigenes Beispiel (nicht im Skript): fleißiger Biber mit 3 Nichtendzuständen, Bestwert 6 Einsen in 14 Schritten (Lin/Rado)
# Eingabealphabet {1}: der Lauf startet auf leerem Band, also mit dem Eingabewort ε.
tm Biber3 {
  alphabet 1
  blank □
  states A B C Z
  start A
  accept Z
  A, □ -> B, 1, R
  A, 1 -> Z, 1, R
  B, □ -> C, □, R
  B, 1 -> B, 1, R
  C, □ -> C, 1, L
  C, 1 -> A, 1, L
  layout A@0,0 B@240,0 C@480,0 Z@720,0
}
